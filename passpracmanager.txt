using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;
using System.Data;
using System.Data.SqlClient;
using System.Configuration;

namespace boomrang
{
    public partial class WebForm1 : System.Web.UI.Page
    {
        string connStr = ConfigurationManager.ConnectionStrings["DBConnection"].ConnectionString;

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                LoadData();
            }
        }

        protected void btnSave_Click(object sender, EventArgs e)
        {
            using (SqlConnection conn = new SqlConnection(connStr))
            {
                conn.Open();  // 🔑 Connection explicitly opened

                SqlCommand cmd = new SqlCommand(
                    "INSERT INTO Users(Name, Email) VALUES(@Name, @Email)", conn);

                cmd.Parameters.AddWithValue("@Name", txtName.Text);
                cmd.Parameters.AddWithValue("@Email", txtEmail.Text);

                cmd.ExecuteNonQuery();

                conn.Close(); // 🔑 Connection explicitly closed
            }

            LoadData();
        }

        private void LoadData()
        {
            using (SqlConnection conn = new SqlConnection(connStr))
            {
                conn.Open();

                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Users", conn);
                DataTable dt = new DataTable();
                da.Fill(dt);

                GridView1.DataSource = dt;
                GridView1.DataBind();

                conn.Close();
            }
        }
    }
}

using System;
using System.Data;
using System.Data.SqlClient;

namespace WebApplication7
{
    public partial class WebForm1 : System.Web.UI.Page
    {
        // Only DataSet in ViewState (SAFE)
        private DataSet ds
        {
            get { return ViewState["ds"] as DataSet; }
            set { ViewState["ds"] = value; }
        }

        string connStr = @"Data Source=(localdb)\MSSQLLocalDB;Initial Catalog=Employees;Integrated Security=True";

        protected void btnLoad_Click(object sender, EventArgs e)
        {
            using (SqlConnection con = new SqlConnection(connStr))
            {
                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Employees", con);

                DataSet temp = new DataSet();
                da.Fill(temp, "Employees");

                ds = temp; // store ONLY dataset

                GridView1.DataSource = ds.Tables["Employees"];
                GridView1.DataBind();
            }
        }

        protected void btnUpdate_Click(object sender, EventArgs e)
        {
            if (ds == null)
            {
                Response.Write("Click Load first!");
                return;
            }

            using (SqlConnection con = new SqlConnection(connStr))
            {
                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Employees", con);

                SqlCommandBuilder cb = new SqlCommandBuilder(da);

                // Modify data in memory
                if (ds.Tables["Employees"].Rows.Count > 0)
                {
                    ds.Tables["Employees"].Rows[0]["Name"] = "Updated Name";
                }

                // Push update to DB
                da.Update(ds, "Employees");

                // Refresh Grid
                GridView1.DataSource = ds.Tables["Employees"];
                GridView1.DataBind();
            }
        }
    }
}

using System;
using System.Configuration;
using System.Data;
using System.Data.SqlClient;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace charizard
{
    public partial class WebForm1 : System.Web.UI.Page
    {
        string cs = ConfigurationManager.ConnectionStrings["DemoConnection"].ConnectionString;

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                BindGrid();
                BindDropdown();
            }
        }

        void BindGrid()
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Students", con);
                DataTable dt = new DataTable();
                da.Fill(dt);

                GridView1.DataSource = dt;
                GridView1.DataBind();

                // Show first record initially
                DetailsView1.DataSource = dt;
                DetailsView1.DataBind();
            }
        }

        void BindDropdown()
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Students", con);
                DataTable dt = new DataTable();
                da.Fill(dt);

                DropDownList1.DataSource = dt;
                DropDownList1.DataTextField = "Name";
                DropDownList1.DataValueField = "Id";
                DropDownList1.DataBind();
            }
        }

        // GridView selection → updates DetailsView
        protected void GridView1_SelectedIndexChanged(object sender, EventArgs e)
        {
            int index = GridView1.SelectedIndex;

            if (index >= 0)
            {
                DetailsView1.PageIndex = index;
            }
        }

        // 🔥 NEW: Dropdown selection → updates DetailsView
        protected void DropDownList1_SelectedIndexChanged(object sender, EventArgs e)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                string query = "SELECT * FROM Students WHERE Id = @Id";
                SqlDataAdapter da = new SqlDataAdapter(query, con);
                da.SelectCommand.Parameters.AddWithValue("@Id", DropDownList1.SelectedValue);

                DataTable dt = new DataTable();
                da.Fill(dt);

                DetailsView1.DataSource = dt;
                DetailsView1.DataBind();
            }
        }
    }
}

                SqlCommand cmd = new SqlCommand("InsertStudent", con);
                cmd.CommandType = CommandType.StoredProcedure;

                cmd.Parameters.AddWithValue("@Name", txtName.Text);
                cmd.Parameters.AddWithValue("@Marks", txtMarks.Text);

                con.Open();
                cmd.ExecuteNonQuery();
                con.Close();
            }

            LoadData();
        }

        private void LoadData()
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlDataAdapter da = new SqlDataAdapter("SELECT * FROM Students1", con);
                DataTable dt = new DataTable();
                da.Fill(dt);

                GridView1.DataSource = dt;
                GridView1.DataBind();
            }
        }

        [HttpPost]
        public ActionResult Edit(Employee e)
        {
            db.Entry(e).State = System.Data.Entity.EntityState.Modified;
            db.SaveChanges();
            return RedirectToAction("Index");
        }

        {
            var emp = db.Employees.Find(id);
            return View(emp);
        }

        public ActionResult DeleteConfirmed(int id)
        {

            return RedirectToAction("Index");
